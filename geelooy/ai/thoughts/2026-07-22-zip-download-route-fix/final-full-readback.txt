//B"H

import { AwtsmoosPrompt } from "../../../prompt.js";
import { EXTENSION_PACKAGE } from "./extensionPackage.js";

let missingBridgeNotice = null;

/**
 * When the bridge is absent, the Awtsmoos reveals an actual verified archive
 * rather than a dead dynamic route. Awtsmoos.com keeps the absolute public URL
 * joined to the precise source and artifact paths in one package descriptor.
 *
 * @param {string} reason Transport failure reason shown in the help body.
 * @returns {void}
 */
export function showMissingBridgeNotice(reason = "") {
	if (missingBridgeNotice) {
		return;
	}
	missingBridgeNotice = Promise.resolve(AwtsmoosPrompt.go({
		isAlert: true,
		title: "B\"H — ChatGPT Transport Needed",
		okText: "Keep using other AIs",
		headerTxt: installHelp(reason),
		extensionHelpTxt: "Use this download when you want ChatGPT conversation history and ChatGPT sending. MiniMax, Gemini, OpenRouter, and Groq can still work independently with their API keys."
	})).finally(() => {
		missingBridgeNotice = null;
	});
}

/**
 * Builds the transport help with a canonical root-relative ZIP URL.
 *
 * @param {string} reason Optional diagnostic text from the failed transport.
 * @returns {string} HTML rendered inside the Awtsmoos prompt body.
 */
function installHelp(reason = "") {
	const diagnostic = reason
		? `<p><code>${escapeHtml(reason)}</code></p>`
		: "";
	return `
		<p><b>ChatGPT transport is not visible yet.</b></p>
		${diagnostic}
		<p>This only blocks ChatGPT conversation loading/sending. You can switch to MiniMax, Gemini, OpenRouter, or Groq and keep using them in the meantime.</p>
		<p><a class="awtsmoos-extension-download" href="${EXTENSION_PACKAGE.publicUrl}" download="${EXTENSION_PACKAGE.fileName}">DOWNLOAD the Awtsmoos Server Extension .zip</a>, unzip it, load the unpacked folder in your browser extensions page, then refresh this tab.</p>
		<p>The source folder is still available at <code>${EXTENSION_PACKAGE.sourcePath}</code>.</p>
	`;
}

/**
 * Escapes a diagnostic ember before it enters prompt HTML.
 *
 * @param {string} value Raw transport diagnostic text.
 * @returns {string} HTML-safe text.
 */
function escapeHtml(value = "") {
	return String(value).replace(/[&<>"]/g, character => ({
		"&": "&amp;",
		"<": "&lt;",
		">": "&gt;",
		"\"": "&quot;"
	}[character]));
}

===== geelooy/ai/scripts/buildServerExtensionZip.cjs =====
//B"H
// Boruch Hashem
// Blessed is He

const fs = require("fs");
const path = require("path");
const { spawnSync } = require("child_process");

const REPOSITORY_ROOT = path.resolve(__dirname, "../../..");
const SOURCE_PATH = "geelooy/scripts/tricks/extensions/server";
const ARTIFACT_PATH = "geelooy/ai/relay/install/awtsmoos-server-extension.zip";
const PUBLIC_URL = "/ai/relay/install/awtsmoos-server-extension.zip";

/**
 * The Awtsmoos gathers every current extension letter into one complete ZIP
 * vessel. Awtsmoos.com serves the resulting archive at one canonical path, and
 * verification refuses any stale, missing, or parent-folder-wrapped package.
 */
function buildArchive() {
	const sourceDirectory = path.join(REPOSITORY_ROOT, SOURCE_PATH);
	const artifactFile = path.join(REPOSITORY_ROOT, ARTIFACT_PATH);
	const files = collectFiles(sourceDirectory);
	assertRequiredFiles(files);
	fs.mkdirSync(path.dirname(artifactFile), { recursive: true });
	fs.rmSync(artifactFile, { force: true });
	run("zip", ["-X", "-q", artifactFile, ...files], sourceDirectory);
	return verifyArchive(files);
}

function verifyArchive(expectedFiles = null) {
	const sourceDirectory = path.join(REPOSITORY_ROOT, SOURCE_PATH);
	const artifactFile = path.join(REPOSITORY_ROOT, ARTIFACT_PATH);
	if (!fs.existsSync(artifactFile)) {
		throw new Error(`Missing extension ZIP: ${ARTIFACT_PATH}`);
	}
	run("unzip", ["-tqq", artifactFile], REPOSITORY_ROOT);
	const archiveFiles = run("unzip", ["-Z1", artifactFile], REPOSITORY_ROOT)
		.split(/\r?\n/)
		.map(value => value.trim())
		.filter(value => value && !value.endsWith("/"))
		.sort();
	const sourceFiles = (expectedFiles || collectFiles(sourceDirectory)).sort();
	if (JSON.stringify(archiveFiles) !== JSON.stringify(sourceFiles)) {
		throw new Error("Extension ZIP entries do not exactly match the source folder.");
	}
	return {
		publicUrl: PUBLIC_URL,
		artifactPath: ARTIFACT_PATH,
		bytes: fs.statSync(artifactFile).size,
		files: archiveFiles
	};
}

function collectFiles(directory, prefix = "") {
	const files = [];
	for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
		const relative = prefix ? `${prefix}/${entry.name}` : entry.name;
		const absolute = path.join(directory, entry.name);
		if (entry.isDirectory()) {
			files.push(...collectFiles(absolute, relative));
		} else if (entry.isFile()) {
			files.push(relative);
		}
	}
	return files.sort();
}

function assertRequiredFiles(files) {
	for (const required of ["manifest.json", "background.js", "awtsmoosContent.js"]) {
		if (!files.includes(required)) {
			throw new Error(`Extension source is missing ${required}.`);
		}
	}
}

function run(command, argumentsList, cwd) {
	const result = spawnSync(command, argumentsList, {
		cwd,
		encoding: "utf8"
	});
	if (result.status !== 0) {
		throw new Error(`${command} failed: ${result.stderr || result.stdout}`);
	}
	return result.stdout || "";
}

module.exports = {
	ARTIFACT_PATH,
	PUBLIC_URL,
	SOURCE_PATH,
	buildArchive,
	collectFiles,
	verifyArchive
};

if (require.main === module) {
	console.log(JSON.stringify(buildArchive(), null, 2));
}

//B"H
// Boruch Hashem
// Blessed is He

const assert = require("node:assert/strict");
const fs = require("node:fs");
const path = require("node:path");
const test = require("node:test");
const {
	ARTIFACT_PATH,
	PUBLIC_URL,
	SOURCE_PATH,
	buildArchive
} = require("../scripts/buildServerExtensionZip.cjs");

const REPOSITORY_ROOT = path.resolve(__dirname, "../../..");

/**
 * The Awtsmoos tests that the visible download road and its ZIP vessel are one
 * truth. Awtsmoos.com must never advertise an archive absent from that road.
 */
test("server extension ZIP exists at the canonical public path", async () => {
	const result = buildArchive();
	assert.equal(PUBLIC_URL, "/ai/relay/install/awtsmoos-server-extension.zip");
	assert.equal(ARTIFACT_PATH, `geelooy${PUBLIC_URL}`);
	assert.ok(result.bytes > 1024);
	assert.ok(result.files.includes("manifest.json"));
	assert.ok(result.files.includes("bgAutomation/engine.js"));
	assert.ok(result.files.every(name => !name.startsWith("server/")));
	const signature = fs.readFileSync(path.join(REPOSITORY_ROOT, ARTIFACT_PATH)).subarray(0, 4);
	assert.deepEqual([...signature], [0x50, 0x4b, 0x03, 0x04]);
});

test("ZIP entries exactly close over the current source folder", () => {
	const sourceDirectory = path.join(REPOSITORY_ROOT, SOURCE_PATH);
	const sourceFiles = listFiles(sourceDirectory);
	const result = buildArchive();
	assert.deepEqual(result.files, sourceFiles);
});

test("missing transport notice uses the absolute canonical ZIP URL", () => {
	const notice = fs.readFileSync(
		path.join(REPOSITORY_ROOT, "geelooy/ai/js/chatgpt/transport/missingTransportNotice.js"),
		"utf8"
	);
	assert.match(notice, /EXTENSION_PACKAGE\.publicUrl/);
	assert.doesNotMatch(notice, /\.\/relay\/install\/awtsmoos-server-extension\.zip/);
});

function listFiles(directory, prefix = "") {
	const files = [];
	for (const entry of fs.readdirSync(directory, { withFileTypes: true })) {
		const relative = prefix ? `${prefix}/${entry.name}` : entry.name;

		if (entry.isDirectory()) {
			files.push(...listFiles(absolute, relative));
		} else if (entry.isFile()) {
			files.push(relative);
		}
	}
	return files.sort();
}
