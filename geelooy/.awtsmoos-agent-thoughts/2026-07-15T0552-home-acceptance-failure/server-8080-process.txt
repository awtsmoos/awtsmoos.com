#!/bin/bash
# B"H
# Boruch Hashem
# Blessed is He

# One persistent supervisor recreates production whenever its child exits. The
# Awtsmoos accepts each new generation only after exactly two canonical shards,
# both persisted indexes, the shared library road, and real hydration are warmed.

set -u

REPOSITORY='/Users/awtsmoos/Documents/awtsmoos/git/awtsmoos.com'
REVIEW='/Users/awtsmoos/Documents/dayuhChadash-review/awtsmoosdb-reclaim-20260712T010855-0700'
NODE_BINARY='/Users/awtsmoos/.nvm/versions/node/v24.17.0/bin/node'
PYTHON_BINARY='/usr/local/bin/python3'
PROBE="$REVIEW/tools/warm_rag_readiness.py"
WARM_REPORT="$REVIEW/evidence/managed-production-warmup-current.json"
WARM_LOG="$REVIEW/evidence/managed-production-warmup-current.log"
CHILD_PID_FILE="$REPOSITORY/.logs/awtsmoos-production-child.pid"
SUPERVISOR_PID_FILE="$REPOSITORY/.logs/awtsmoos-production-supervisor.pid"
CHILD_PID=''
STOP_REQUESTED=false

terminate_child() {
	if [ -z "$CHILD_PID" ]; then return; fi
	if ! kill -0 "$CHILD_PID" 2>/dev/null; then return; fi
	kill -TERM "$CHILD_PID" 2>/dev/null || true
	for _ in $(seq 1 40); do
		if ! kill -0 "$CHILD_PID" 2>/dev/null; then return; fi
		sleep 0.25
	done
	kill -KILL "$CHILD_PID" 2>/dev/null || true
}

request_stop() {
	STOP_REQUESTED=true
	terminate_child
}

wait_for_listener() {
	for _ in $(seq 1 160); do
		if lsof -nP -iTCP:8080 -sTCP:LISTEN >/dev/null 2>&1; then
			return 0
		fi
		if ! kill -0 "$CHILD_PID" 2>/dev/null; then
			return 1
		fi
		sleep 0.25
	done
	return 1
}

warm_production() {
	for attempt in $(seq 1 3); do
		rm -f "$WARM_REPORT"
		if "$PYTHON_BINARY" "$PROBE" \
			'http://127.0.0.1:8080/api/social' \
			"$WARM_REPORT" \
			> "$WARM_LOG" 2>&1; then
			echo "B\"H six-stage production readiness passed on attempt $attempt"
			return 0
		fi
		sleep 1
	done
	return 1
}

run_child_cycle() {
	cd "$REPOSITORY" || return 1
	export AWTSMOOS_RAG_EF_SEARCH='96'
	"$NODE_BINARY" index.js &
	CHILD_PID=$!
	echo "$CHILD_PID" > "$CHILD_PID_FILE"
	echo "B\"H production child PID $CHILD_PID"
	if ! wait_for_listener; then
		terminate_child
		return 1
	fi
	if ! warm_production; then
		terminate_child
		return 1
	fi
	wait "$CHILD_PID"
	status=$?
	CHILD_PID=''
	rm -f "$CHILD_PID_FILE"
	return "$status"
}

trap request_stop TERM INT
mkdir -p "$REPOSITORY/.logs"
echo "$$" > "$SUPERVISOR_PID_FILE"
while [ "$STOP_REQUESTED" = false ]; do
	run_child_cycle || true
	if [ "$STOP_REQUESTED" = false ]; then
		echo 'B"H production child ended; recreating after two seconds'
		sleep 2
	fi
done
rm -f "$CHILD_PID_FILE" "$SUPERVISOR_PID_FILE"
exit 0
